'use client';

import { useEffect, useState } from 'react';
import Link from 'next/link';
import api from '@/api/client';

export default function Dashboard() {
  const [counts, setCounts] = useState({ menu: 0, lowStock: 0, pendingPOs: 0 });

  useEffect(() => {
    let cancelled = false;
    (async () => {
      const [menu, low, pos] = await Promise.all([
        api.get('/menu'),
        api.get('/inventory/low-stock'),
        api.get('/purchase-orders'),
      ]);
      if (cancelled) return;
      setCounts({
        menu: menu.data.length,
        lowStock: low.data.length,
        pendingPOs: pos.data.filter((p) => p.status === 'pending_supplier').length,
      });
    })().catch(console.error);
    return () => { cancelled = true; };
  }, []);

  return (
    <div>
      <h1>Dashboard</h1>
      <p className="muted">One restaurant, one kitchen. Five workflows.</p>
      <div className="grid">
        <Link href="/menu" className="card stat">
          <span className="stat-num">{counts.menu}</span>
          <span className="stat-label">Menu items</span>
        </Link>
        <Link href="/inventory" className="card stat">
          <span className="stat-num warn">{counts.lowStock}</span>
          <span className="stat-label">Low-stock materials</span>
        </Link>
        <Link href="/purchase-orders" className="card stat">
          <span className="stat-num">{counts.pendingPOs}</span>
          <span className="stat-label">Pending POs</span>
        </Link>
        <Link href="/orders" className="card stat">
          <span className="stat-num">â†’</span>

  const [error, setError] = useState('');
  const [busy, setBusy] = useState(false);

  async function load() {
    try {
      const [p, inv] = await Promise.all([api.get('/purchase-orders'), api.get('/inventory')]);
      setPOs(p.data);
      setMaterials(inv.data);
      if (inv.data.length && !inv.data.some((m) => m.material_id === form.material_id)) {
        setForm((f) => ({ ...f, material_id: inv.data[0].material_id, unit: inv.data[0].unit }));
      }
    } catch (err) { setError(err.response?.data?.error || err.message); }
  }
  useEffect(() => { load(); }, []); // eslint-disable-line react-hooks/exhaustive-deps

  async function create(e) {
    e.preventDefault();
    setBusy(true); setError('');
    try {
      await api.post('/purchase-orders', { ...form, qty: Number(form.qty) });
      await load();
    } catch (err) { setError(err.response?.data?.error || err.message); }
    finally { setBusy(false); }

'use client';
import { useEffect, useState } from 'react';
import api from '../api/client';

export default function Menu() {
  const [items, setItems] = useState([]);
  const [error, setError] = useState('');

  async function load() {
    try {
      const { data } = await api.get('/menu');
      // Guard: if backend returns {items:[â€¦]}, {data:[â€¦]}, or a non-array,
      // extract the array instead of feeding the raw payload into .map.
      const list = Array.isArray(data)
        ? data
        : Array.isArray(data?.items)
          ? data.items
          : Array.isArray(data?.data)
            ? data.data
            : [];
      setItems(list);
    } catch (err) {
      setError(err.response?.data?.error || err.message);
      setItems([]);  // never leave a non-array in state
    }
  }
  useEffect(() => { load(); }, []);

  async function toggle(item) {
    try {
      await api.patch(`/menu/${item.item_id}/availability`, { in_stock: !item.in_stock });
      setItems((prev) =>
        (Array.isArray(prev) ? prev : []).map((i) =>
          i.item_id === item.item_id ? { ...i, in_stock: !i.in_stock } : i
        )
      );
    } catch (err) {
      alert(err.response?.data?.error || err.message);
    }
  }

  return (
    <div>
      <h1>Menu</h1>
      <p className="muted">Flip availability to run <strong>Workflow 1</strong>.</p>
      {error && <div className="alert error">{error}</div>}
      <table className="table">
        <thead>
          <tr>
            <th>Item</th><th>Category</th><th>Price</th><th>Recipe</th><th>In stock</th><th></th>
          </tr>
        </thead>
        <tbody>
          {items.map((i) => (
            <tr key={i.item_id}>
              <td><strong>{i.name}</strong><br/><span className="mono muted">{i.item_id}</span></td>
              <td>{i.category}</td>
              <td>â‚¹{Number(i.price).toFixed(2)}</td>
              <td className="mono small">

              </td>
              <td>
                <span className={i.in_stock ? 'pill ok' : 'pill off'}>
                  {i.in_stock ? 'In stock' : 'Out'}
                </span>

}

'use client';

import { useEffect, useState } from 'react';
import api from '../api/client';

export default function Inventory() {
  const [items, setItems] = useState([]);
  const [error, setError] = useState('');

  async function load() {
    try {
      const { data } = await api.get('/inventory');
      const list = Array.isArray(data)
        ? data
        : Array.isArray(data?.items)
          ? data.items
          : Array.isArray(data?.data)
            ? data.data
            : [];
      setItems(list);
    } catch (err) {
      setError(err.response?.data?.error || err.message);
      setItems([]);
    }
  }
  useEffect(() => { load(); }, []);

  async function adjust(item) {
    const raw = prompt(`Adjust ${item.name} by (e.g. -2 or 5):`, '1');
    if (raw === null) return;
    const n = Number(raw);
    if (!Number.isFinite(n)) return alert('Not a number');
    try {
      const { data } = await api.patch(`/inventory/${item.material_id}`, {
        adjust_qty: n, reason: 'manual adjust from UI',
      });
      setItems((prev) =>
        (Array.isArray(prev) ? prev : []).map((i) =>
          i.material_id === item.material_id
            ? { ...i, current_qty: data.new_qty }
            : i
        )
      );
    } catch (err) {
      alert(err.response?.data?.error || err.message);
    }
  }

  return (
    <div>
      <h1>Inventory</h1>
      <p className="muted">Materials below threshold are highlighted. Adjust quantity to correct stock manually.</p>
      {error && <div className="alert error">{error}</div>}
      <table className="table">
        <thead>
          <tr><th>Material</th><th>Qty</th><th>Unit</th><th>Low threshold</th><th></th></tr>
        </thead>
        <tbody>
          {items.map((i) => {
            const low = i.current_qty < i.low_threshold;
            return (
              <tr key={i.material_id} className={low ? 'row-warn' : ''}>
                <td><strong>{i.name}</strong><br/><span className="mono muted">{i.material_id}</span></td>
                <td className={low ? 'warn' : ''}>{i.current_qty}</td>
                <td>{i.unit}</td>
                <td>{i.low_threshold}</td>
                <td><button className="btn btn-small" onClick={() => adjust(i)}>Adjust</button></td>
              </tr>
            );
          })}
          {items.length === 0 && !error && (
            <tr><td colSpan="5" className="muted">No inventory loaded.</td></tr>
          )}
        </tbody>
      </table>
    </div>
  );
}

'use client';

import { useEffect, useState, useRef } from 'react';
import api from '../api/client';

export default function RunningTables() {
  const [tables, setTables] = useState([]);
  const [activeId, setActiveId] = useState(null);
  const [messages, setMessages] = useState([]);   // chat transcript for the active table
  const [input, setInput] = useState('');
  const [busy, setBusy] = useState(false);
  const [err, setErr] = useState('');
  const endRef = useRef(null);

  async function loadTables() {
    try {
      const { data } = await api.get('/tables');
      setTables(data);
      if (data.length && !activeId) setActiveId(data[0].table_id);
    } catch (e) {
      setErr(e.response?.data?.error || e.message);
    }
  }
  useEffect(() => { loadTables(); }, []); // eslint-disable-line react-hooks/exhaustive-deps
  useEffect(() => { endRef.current?.scrollIntoView({ behavior: 'smooth' }); }, [messages]);

  const active = tables.find((t) => t.table_id === activeId) || null;

  async function handleSend(e) {
    e.preventDefault();
    if (!input.trim() || !active) return;
    const text = input.trim();
    setInput('');
    setMessages((m) => [...m, { role: 'user', text }]);
    setBusy(true); setErr('');

    try {

                </div>

                <span className="mono">â‚¹{active.running_total.toFixed(2)}</span>
              </div>

              <hr />
              <h3>Chat</h3>
              <div className="chat-log">
                {messages.length === 0 && (
                  <p className="muted small">
                    Try: <code>apply the Friends discount to {active.table_id}</code>, or <code>what was excluded?</code>
                  </p>
                )}
                {messages.map((m, i) => (
                  <div key={i} className={`chat-msg ${m.role}`}>
                    <div className="chat-role">{m.role}</div>
                    <pre className="chat-text">{m.text}</pre>
                  </div>
                ))}
                <div ref={endRef} />
              </div>
              <form className="inline-form" onSubmit={handleSend}>
                <input
                  style={{ flex: 1 }}
                  value={input}
                  onChange={(e) => setInput(e.target.value)}
                  placeholder={`Type a commandâ€¦`}
                  disabled={busy}
                />
                <button className="btn btn-primary" disabled={busy || !input.trim()}>
                  {busy ? 'â€¦' : 'Send'}
                </button>
              </form>
            </>
          ) : (
            <p className="muted">Select a table to see its order and chat.</p>
          )}
        </div>
      </div>
    </div>
  );
}

import axios from 'axios';
import { SAGEION_DEMO_APPS } from '@/config/sageionDemos';

export const TOKEN_KEY = 'crm_token';
export const USER_KEY = 'crm_user';

// â­ Resolve the currently active demo id. Set by DemoDetail.jsx's
// "Try Me" button before it navigates to /owner, so by the time any
// API call fires, this reflects the demo the visitor is exploring.
//
// Defaults to 'demo1' when no demo is set (e.g., a direct /owner
// visit, or a fresh page load before "Try Me" has run). That's a
// sensible fallback â€” demo1 is the canonical starting point.
function getActiveDemoId() {
  if (typeof window === 'undefined') return 'demo1';
  try {
    return localStorage.getItem('active_demo') || 'demo1';
  } catch {
    return 'demo1';
  }
}

// â­ Resolve the API base URL from SAGEION_DEMO_APPS[active_demo].
//
// Resolution order:
//   1. active demo's API_BASE_full_URL
//   2. NEXT_PUBLIC_API_BASE_full_URL env var
//   3. any demo's API_BASE_full_URL (guards against a demo entry
//      that omitted the field by mistake)
//   4. hardcoded fallback
//
// Because active_demo can change mid-session (only via a full page
// reload, per our design), this is re-evaluated on every request
// from the interceptor below.
function currentBaseUrl() {
  const activeDemoId = getActiveDemoId();
  const appConfig = SAGEION_DEMO_APPS[activeDemoId];
  if (appConfig?.API_BASE_full_URL) {
    return appConfig.API_BASE_full_URL;
  }

  if (process.env.NEXT_PUBLIC_API_BASE_full_URL) {
    return process.env.NEXT_PUBLIC_API_BASE_full_URL;
  }

  const firstDemo = Object.values(SAGEION_DEMO_APPS)[0];
  if (firstDemo?.API_BASE_full_URL) {
    return firstDemo.API_BASE_full_URL;
  }

  return 'http://122.160.157.99:3001';
}

const api = axios.create({
  // Base URL applied at create time â€” the interceptor re-resolves
  // it per request in case active_demo changes.
  baseURL: currentBaseUrl(),
  headers: { 'Content-Type': 'application/json' },
});

api.interceptors.request.use((config) => {
  // Re-resolve on every request.
  config.baseURL = currentBaseUrl();

  // Attach token
  if (typeof window !== 'undefined') {
    const token = localStorage.getItem(TOKEN_KEY);
    if (token) config.headers.Authorization = `Bearer ${token}`;
  }

  return config;
});

api.interceptors.response.use(
  (r) => r,
  (err) => {
    if (typeof window !== 'undefined' && err.response?.status === 401) {
      localStorage.removeItem(TOKEN_KEY);
      localStorage.removeItem(USER_KEY);
      if (!window.location.pathname.startsWith('/login')) {
        window.location.href = '/login';
      }
    }
    return Promise.reject(err);
  }
);

export function saveSession(data) {
  if (typeof window === 'undefined') return;
  const token = data.token;
  const user_id = data.user_id || data.user?.id;
  const email = data.email || data.user?.email;
  const role = data.role || data.user?.role;

  localStorage.setItem(TOKEN_KEY, token);
  localStorage.setItem(USER_KEY, JSON.stringify({ user_id, email, role }));
}

export function clearSession() {
  if (typeof window === 'undefined') return;
  localStorage.removeItem(TOKEN_KEY);
  localStorage.removeItem(USER_KEY);
}

export function currentUser() {
  if (typeof window === 'undefined') return null;
  try { return JSON.parse(localStorage.getItem(USER_KEY) || 'null'); }
  catch { return null; }
}

export default api;

'use client';

import { createContext, useContext, useMemo, useState, useEffect } from 'react';
import { currentUser, saveSession, clearSession } from './client';
import api from './client';

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  // â­ Distinguishes "no user" (logged out) from "not hydrated yet".
  const [loaded, setLoaded] = useState(false);

  useEffect(() => {
    setUser(currentUser());
    setLoaded(true); // â† hydration complete
  }, []);

  async function login(email, password) {
    const { data } = await api.post('/auth/login', { email, password });
    saveSession(data);
    setUser({ user_id: data.user_id, email: data.email, role: data.role });
    return data;
  }

  // â­ NEW: Request OTP for signup (used by Register.jsx)
  async function sendOtp(email) {
    const { data } = await api.post('/auth/send-otp', { email, purpose: 'signup' });
    return data;
  }

  // â­ UPDATED: Register now requires the OTP (used by VerifyOtp.jsx)
  async function register(email, password, otp) {
    const { data } = await api.post('/auth/register', { email, password, otp });
    saveSession(data);
    setUser({ user_id: data.user_id, email: data.email, role: data.role });
    return data;
  }

  function logout() {
    clearSession();
    setUser(null);
  }

  const value = useMemo(
    () => ({ user, loaded, login, register, sendOtp, logout }),
    [user, loaded],
  );

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export function useAuth() {
  return useContext(AuthContext);
}

import { useEffect } from 'react';

// Module-scope set of appIds we've bootstrapped on this page load.
// Prevents duplicate <script> tags and duplicate setUp() calls if
// the component remounts for the same app.
const BOOTSTRAPPED_APP_IDS = new Set();

export default function SageionScripts({ appConfig }) {
  useEffect(() => {
    const appId = appConfig?.SAGEION_APP_ID || appConfig?.appId;
    const apiKey = appConfig?.SAGEION_API_KEY || appConfig?.apiKey;
    const region = appConfig?.SAGEION_REGION || appConfig?.region || 'US';
    const rootId = appConfig?.ROOT_SAGEION_ID || appConfig?.rootId || '';

    if (!appId || !apiKey) {
      console.error(
        '[SageionScripts] missing SAGEION_APP_ID or SAGEION_API_KEY in appConfig',
        appConfig,
      );
      return;
    }

    // Idempotency guards.
    if (BOOTSTRAPPED_APP_IDS.has(appId)) {
      console.log('[SageionScripts] already bootstrapped for', appId);
      return;
    }
    if (BOOTSTRAPPED_APP_IDS.size > 0) {
      console.error(
        '[SageionScripts] a different Sageion app is already bootstrapped',
        Array.from(BOOTSTRAPPED_APP_IDS),
        'â€” refusing to hot-swap to',
        appId,
        '. A full page reload is required to switch apps.',
      );
      return;
    }
    BOOTSTRAPPED_APP_IDS.add(appId);

    // â­ Font Awesome stylesheet â€” added idempotently so a
    // remount doesn't stack duplicate <link> tags.
    if (!document.getElementById('sageion-font-awesome')) {
      const faLink = document.createElement('link');
      faLink.id = 'sageion-font-awesome';
      faLink.rel = 'stylesheet';
      faLink.href =
        'https://cdnjs.cloudflare.com/ajax/libs/font-awesome/6.0.0-beta3/css/all.min.css';
      document.head.appendChild(faLink);
    }

    const isDev = process.env.NODE_ENV === 'development';
    const bundleSrc = isDev
      ? '/local-bundle/bundle.js'
      : 'https://magicchat-core.github.io/prod-ssc-client-cdns/bundle.js';

    // 1. socket.io
    const socketScript = document.createElement('script');
    socketScript.src = 'https://cdn.socket.io/4.1.2/socket.io.min.js';
    socketScript.async = false;
    document.body.appendChild(socketScript);

    // 2. bundle
    const bundleScript = document.createElement('script');
    bundleScript.src = bundleSrc;
    bundleScript.async = false;

    bundleScript.onload = async () => {
      console.log('[SageionScripts] bundle loaded for', appId);

      if (!window.sageion_os) {
        console.error('[SageionScripts] window.sageion_os not present after bundle load');
        return;
      }

      try {
        await window.sageion_os.setUp(appId, apiKey, region, rootId);
        console.log('[SageionScripts] setUp complete for', appId);

        window.dispatchEvent(
          new CustomEvent('sageion_setup_complete', {
            detail: { appId },
          }),
        );
      } catch (err) {
        console.error('[SageionScripts] setUp failed for', appId, err);
      }
    };

    document.body.appendChild(bundleScript);

    // Cleanup: hide the floating bubble when this component unmounts.
    return () => {
      try {
        const bubble = document.querySelector('[id^="chat_modal_opener__"]');
        if (bubble) bubble.style.display = 'none';
        const modal = document.querySelector('[class^="chat_modal__"]');
        if (modal) modal.style.display = 'none';
      } catch (_) {
        /* no-op */
      }
    };
  }, [appConfig]);

  return null;
}

'use client';

import { useEffect, useState } from 'react';
import { useAuth } from '../api/AuthContext';

export default function SageionBridge({ appConfig }) {
  const { user, loaded } = useAuth();
  const [isReady, setIsReady] = useState(false);

  const appId = appConfig?.SAGEION_APP_ID || appConfig?.appId;

  useEffect(() => {
    const handleSetupComplete = (e) => {
      // â­ Filter to OUR app id, in case two bridges ever coexist.
      if (!e.detail || !appId || e.detail.appId === appId) {
        console.log(
          '[SageionBridge] received sageion_setup_complete for',
          e.detail?.appId,
        );
        setIsReady(true);
      }
    };

    window.addEventListener('sageion_setup_complete', handleSetupComplete);

    // Fallback: event may have fired before we mounted.
    if (window.sageion_os && window.sageion_os._isSetupComplete) {
      setIsReady(true);
    }

    return () => {
      window.removeEventListener('sageion_setup_complete', handleSetupComplete);
    };
  }, [appId]);

  // â­ Only depend on user_id, not the whole user object.
  // AuthContext creates a fresh object on every login, so depending
  // on `user` would re-run initialize() even for the same user.
  const userId = user?.user_id;

  useEffect(() => {
    if (!isReady) return;
    if (!loaded) return; // â† don't boot in anonymous mode mid-hydration

    window.sageion_os
      .initialize(userId ? { uid: String(userId) } : {})
      .then(() => {
        // â­ Restore bubble visibility after initialize. SageionScripts
        // hides it on unmount; this flips it back when we re-enter the
        // authed tree.
        const bubble = document.querySelector('[id^="chat_modal_opener__"]');
        if (bubble) bubble.style.display = '';
      })
      .catch((err) => console.error('[Sageion] initialize failed:', err));
  }, [isReady, loaded, userId]);

  return null;
}

'use client';

import { useEffect } from 'react';
import { useRouter } from 'next/navigation';
import { useAuth } from '@/api/AuthContext';

export default function ProtectedRoute({ children }) {
  const { user, loaded } = useAuth();
  const router = useRouter();

  useEffect(() => {
    // â­ Only redirect once auth has actually hydrated. Before `loaded`
    // flips, `user === null` is ambiguous â€” it means "haven't checked
    // localStorage yet", not "logged out".
    if (loaded && !user) {
      router.replace('/login');
    }
  }, [loaded, user, router]);

  // â­ Render nothing while auth is hydrating. One frame in practice.
  if (!loaded) return null;

  // â­ After hydration with no user â€” the redirect effect above is
  // queued. Return null so we don't flash protected content.
  if (!user) return null;

  return children;
}

// app/(app)/layout.jsx
'use client';

import { useEffect, useRef, useState } from 'react';
import ProtectedRoute from '@/components/ProtectedRoute';
import Layout from '@/components/Layout';
import DemoGuide from '@/components/DemoGuide'; 
import { getSageionApp } from '@/config/sageionDemos';
import { useAuth } from '@/api/AuthContext';

export default function AuthedLayout({ children }) {
  const { user, loaded } = useAuth();
  const [activeDemoId, setActiveDemoId] = useState(null);
  const handleRef = useRef(null);

  useEffect(() => {
    setActiveDemoId(localStorage.getItem('active_demo') || 'demo1');
  }, []);

  useEffect(() => {
    if (!activeDemoId || !loaded) return;
    const appConfig = getSageionApp(activeDemoId);
    if (!appConfig) return;

    const handle = window.sageion_os.mount({
      appId: appConfig.SAGEION_APP_ID,
      apiKey: appConfig.SAGEION_API_KEY,
      region: appConfig.SAGEION_REGION,
      rootId: appConfig.ROOT_SAGEION_ID || '',
      match: '/',
    });
    handleRef.current = handle;

    handle.updateUid(user?.user_id ?? null);

    return () => {
      handleRef.current = null;
      handle.unmount();
    };
  }, [activeDemoId, loaded]);

  useEffect(() => {
    if (handleRef.current && loaded) {
      handleRef.current.updateUid(user?.user_id ?? null);
    }
  }, [user?.user_id, loaded]);

  // â­ NEW: Listen for the "Use â†’" button click from DemoGuide
  useEffect(() => {
    const handleInjectPrompt = (e) => {
      const { prompt } = e.detail;
      
      // Sageion mounts to document.body, so we can query it directly.
      // We use a slight delay to ensure the chat input is fully rendered if the chat was just opened.
      setTimeout(() => {
        // Try to find the textarea (Sageion's chat input)
        const chatInput = document.querySelector('textarea[placeholder*="Type a message"]') || 
                          document.querySelector('textarea') || 
                          document.querySelector('input[type="text"]'); 
        
        if (chatInput) {
          chatInput.value = prompt;
          // React inputs need this native event to trigger their state change
          const event = new Event('input', { bubbles: true });
          chatInput.dispatchEvent(event);
          chatInput.focus();
        } else {
          console.warn('[DemoGuide] Could not find Sageion chat input to inject prompt.');
        }
      }, 300); 
    };

    window.addEventListener('sageion_inject_prompt', handleInjectPrompt);
    return () => window.removeEventListener('sageion_inject_prompt', handleInjectPrompt);
  }, []);

  return (
    <ProtectedRoute>
      <Layout>
        {children}
      </Layout>
      <DemoGuide /> 
    </ProtectedRoute>
  );
}
